// Profiler-only (never committed to the game): "-dungeoncheck" logs, every 10 s for 300 s, per ECS world: dungeon
// entities (DungeonSocket buffers), DungeonPost count, creatures (CreatureState) and replicas, and how many of each are
// within 60 m of that world's player. Copied to Assets/Game/Scripts/Debugging/DungeonCheck.cs for one build, then deleted.
// Lines start "DUNGEONCHECK".
using System.Collections;
using Unity.Collections;
using Unity.Entities;
using Unity.Mathematics;
using Unity.Transforms;
using UnityEngine;

public class DungeonCheck : MonoBehaviour
{
    [RuntimeInitializeOnLoadMethod(RuntimeInitializeLoadType.AfterSceneLoad)]
    static void Boot()
    {
        if (System.Array.IndexOf(System.Environment.GetCommandLineArgs(), "-dungeoncheck") < 0 || FindAnyObjectByType<DungeonCheck>() != null)
            return;
        var go = new GameObject("DungeonCheck");
        DontDestroyOnLoad(go);
        go.AddComponent<DungeonCheck>();
    }

    void OnEnable() => StartCoroutine(Run());

    IEnumerator Run()
    {
        for (int k = 1; k <= 30; k++)
        {
            yield return new WaitForSecondsRealtime(10f);
            foreach (var w in World.All)
            {
                if (!w.IsCreated || (w.Flags & WorldFlags.Game) == 0) continue;
                try { Log(w); } catch (System.Exception e) { Debug.Log($"DUNGEONCHECK {w.Name}: {e.GetBaseException().Message}"); }
            }
        }
    }

    static void Log(World w)
    {
        var em = w.EntityManager;
        using var dq = em.CreateEntityQuery(ComponentType.ReadOnly<Punchy.Dungeon.DungeonSocket>());
        using var pq = em.CreateEntityQuery(ComponentType.ReadOnly<Punchy.Fauna.DungeonPost>(), ComponentType.ReadOnly<Punchy.Fauna.CreatureState>());
        using var cq = em.CreateEntityQuery(ComponentType.ReadOnly<Punchy.Fauna.CreatureState>());
        using var rq = em.CreateEntityQuery(ComponentType.ReadOnly<Punchy.Net.CreatureReplica>());
        using var plq = em.CreateEntityQuery(ComponentType.ReadOnly<Punchy.Player.PlayerTag>(), ComponentType.ReadOnly<LocalTransform>());
        var player = float3.zero; var hasPlayer = false;
        using (var pl = plq.ToComponentDataArray<LocalTransform>(Allocator.Temp))
            if (pl.Length > 0) { player = pl[0].Position; hasPlayer = true; }
        int near = 0, postsNear = 0, guards = 0, drinks = 0;
        using (var cs = cq.ToComponentDataArray<Punchy.Fauna.CreatureState>(Allocator.Temp))
            foreach (var c in cs) if (hasPlayer && math.distance(c.Position, player) < 60f) near++;
        using (var ps = pq.ToComponentDataArray<Punchy.Fauna.DungeonPost>(Allocator.Temp))
        using (var pcs = pq.ToComponentDataArray<Punchy.Fauna.CreatureState>(Allocator.Temp))
            for (int i = 0; i < ps.Length; i++)
            {
                if (hasPlayer && math.distance(pcs[i].Position, player) < 60f) postsNear++;
                var b = ps[i].Behaviour.ToString();
                if (b.Contains("Guard")) guards++; else if (b.Contains("Drink")) drinks++;
            }
        // Job 21d: what the creatures at the NearWater / NearChest sockets are doing in this world (on a client these are
        // replicas, so this shows whether the server's drink/guard behaviour reaches it): drinking = in Drink mode within
        // 2 m of a water socket; at a chest = within 3 m of a chest socket; facing = a player within 7 m and the creature
        // turned to within 30 degrees of them. Every player in the world is listed too.
        var water = new System.Collections.Generic.List<float3>(); var chest = new System.Collections.Generic.List<float3>();
        using (var des = dq.ToEntityArray(Allocator.Temp))
            foreach (var de in des)
                foreach (var so in em.GetBuffer<Punchy.Dungeon.DungeonSocket>(de, true))
                {
                    if ((so.Tags & (uint)Punchy.Dungeon.SocketTags.NearWater) != 0) water.Add(so.Pos);
                    if ((so.Tags & (uint)Punchy.Dungeon.SocketTags.NearChest) != 0) chest.Add(so.Pos);
                }
        var players = new System.Collections.Generic.List<float3>();
        using (var pl = plq.ToComponentDataArray<LocalTransform>(Allocator.Temp))
            foreach (var t in pl) players.Add(t.Position);
        int atWater = 0, drinking = 0, atChest = 0, facing = 0;
        var detail = new System.Text.StringBuilder();
        using (var cs = cq.ToComponentDataArray<Punchy.Fauna.CreatureState>(Allocator.Temp))
            foreach (var c in cs)
            {
                foreach (var wp in water)
                    if (math.distance(c.Position.xz, wp.xz) < 2f && math.abs(c.Position.y - wp.y) < 4f)
                    {
                        atWater++;
                        if (c.Mode == Punchy.Fauna.CreatureMode.Drink) drinking++;
                        detail.Append($" water({wp.x:F0},{wp.z:F0}):{c.Mode}/{c.Stage}");
                        break;
                    }
                foreach (var cp in chest)
                    if (math.distance(c.Position.xz, cp.xz) < 3f && math.abs(c.Position.y - cp.y) < 4f)
                    {
                        atChest++;
                        foreach (var pp in players)
                        {
                            var to = pp.xz - c.Position.xz;
                            if (math.length(to) > 7f) continue;
                            var off = math.abs(math.atan2(math.sin(math.atan2(to.x, to.y) - c.Yaw), math.cos(math.atan2(to.x, to.y) - c.Yaw)));
                            if (off < math.radians(30f)) { facing++; break; }
                        }
                        detail.Append($" chest({cp.x:F0},{cp.y:F0},{cp.z:F0}):{c.Mode}");
                        break;
                    }
            }
        var who = new System.Text.StringBuilder();
        foreach (var pp in players) who.Append($" ({pp.x:F0},{pp.y:F0},{pp.z:F0})");
        Debug.Log($"DUNGEONPOSTS t {Time.realtimeSinceStartup:F0} {w.Name}: water sockets {water.Count}, chest sockets {chest.Count}; at water {atWater} (drinking {drinking}), at a chest {atChest} (facing a player {facing}); players {players.Count}:{who} |{detail}");
        Debug.Log($"DUNGEONCHECK t {Time.realtimeSinceStartup:F0} {w.Name}: dungeons {dq.CalculateEntityCount()}, posts {pq.CalculateEntityCount()} (guard {guards}, drink {drinks}, {postsNear} within 60 m), " +
                  $"creatures {cq.CalculateEntityCount()} ({near} within 60 m), replicas {rq.CalculateEntityCount()}, player {(hasPlayer ? $"({player.x:F0}, {player.y:F0}, {player.z:F0})" : "none")}");
    }
}
